# Redirige todos mis dominios a https://1d-solutions.com
RewriteEngine On
RewriteCond %{HTTPS} off
RewriteRule ^(.*)$ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# Protege el archivo htaccess
<files .htaccess>
order allow,deny
deny from all
</files>

<IfModule mod_expires.c>
# Enable expirations
ExpiresActive On
# Default directive
ExpiresDefault "access plus 1 days"
ExpiresByType application/x-msdownload "access 6 hour"
</IfModule>

<ifModule mod_headers.c>
    Header set Vary "Accept-Encoding"
</ifModule>

<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/css text/html text/javascript text/plain text/xml
    AddOutputFilterByType DEFLATE font/opentype font/otf font/ttf image/svg+xml image/x-icon
    AddOutputFilterByType DEFLATE application/vnd.ms-fontobject application/x-font-ttf
    AddOutputFilterByType DEFLATE application/x-font application/x-font-truetype
    AddOutputFilterByType DEFLATE application/x-font-opentype application/x-font-otf
    AddOutputFilterByType DEFLATE application/javascript application/x-javascript
    AddOutputFilterByType DEFLATE application/xhtml+xml application/xml application/rss+xml
</IfModule>


# ▼▼▼ INICIO AÑADIDO 2026-08-15 — 1. Cabeceras de seguridad ▼▼▼
#
# Sólo añaden cabeceras a la respuesta: no cambian contenido ni rutas. Van en su
# propio <IfModule> para que el bloque se pueda borrar de una pieza.
#
# vault: postura-de-seguridad-del-sitio · decisión 0003
#
<IfModule mod_headers.c>

  # HTTPS obligatorio durante un año. Sin includeSubDomains (dejaría inaccesible
  # un subdominio futuro sin certificado) y sin preload (es casi irreversible).
  # Única directiva del archivo con efecto persistente en el navegador: quien la
  # reciba usará HTTPS un año aunque mañana se quite.
  Header always set Strict-Transport-Security "max-age=31536000"

  # El navegador no adivina el tipo del archivo: respeta el Content-Type.
  Header always set X-Content-Type-Options "nosniff"

  # Sólo el propio sitio puede embeberlo en un iframe (clickjacking).
  Header always set X-Frame-Options "SAMEORIGIN"

  # Hacia otro dominio se envía sólo el origen; dentro del sitio, la URL
  # completa, que es lo que Analytics necesita para el informe de referencias.
  Header always set Referrer-Policy "strict-origin-when-cross-origin"

  # El sitio no usa estas APIs. Declararlas vacías impide que las pida un script
  # inyectado o una etiqueta nueva de GTM.
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=(), usb=()"

  # Oculta la versión de PHP si el hosting la anuncia.
  Header always unset X-Powered-By

</IfModule>
#
# ▲▲▲ FIN AÑADIDO — 1. Cabeceras de seguridad ▲▲▲


# ▼▼▼ INICIO AÑADIDO 2026-08-15 — 2. Content-Security-Policy (MODO INFORME) ▼▼▼
#
# "Report-Only" no bloquea: sólo registra las violaciones en la consola del
# navegador, así que no puede romper el sitio. Lo que aporta es la lista de
# dominios permitidos. Sigue en modo informe porque pasarlo a bloqueo exige
# decidir qué hacer con el 'unsafe-eval' que GTM necesita.
#
# LA CABECERA VA EN UNA SOLA LÍNEA. Apache admite partirla con "\", pero si
# queda un espacio detrás de la barra el sitio devuelve 500. No la partas.
#
# vault: decisión 0003 · postura-de-seguridad-del-sitio
#
<IfModule mod_headers.c>
  Header always set Content-Security-Policy-Report-Only "default-src 'self'; script-src 'self' 'unsafe-inline' https://www.googletagmanager.com https://www.google-analytics.com https://googleads.g.doubleclick.net https://www.googleadservices.com https://www.google.com https://web3forms.com https://*.hcaptcha.com https://hcaptcha.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com https://*.hcaptcha.com; font-src 'self' https://fonts.gstatic.com data:; img-src 'self' data: https://www.googletagmanager.com https://www.google-analytics.com https://www.google.com https://googleads.g.doubleclick.net https://td.doubleclick.net https://pagead2.googlesyndication.com; connect-src 'self' https://www.google-analytics.com https://region1.google-analytics.com https://analytics.google.com https://www.googletagmanager.com https://api.web3forms.com https://pagead2.googlesyndication.com https://*.hcaptcha.com; frame-src https://www.googletagmanager.com https://td.doubleclick.net https://*.hcaptcha.com https://hcaptcha.com; form-action 'self' https://api.web3forms.com https://store.payproglobal.com; base-uri 'self'; object-src 'none'; frame-ancestors 'self'"
</IfModule>

# ▼▼▼ INICIO AÑADIDO 2026-08-15 — 3. Redirección de las páginas de /1dnest/ ▼▼▼
#
# Sub-sitio retirado de 2021-2024: llevaba analítica muerta y enlaces a ShareIt,
# la pasarela anterior a PayPro, que ya no cobra. Se redirige en vez de borrar
# para conservar el enlazado externo. Contenido en español → destino /es/.
#
# vault: sitio-zombi-1dnest · t0013 (borrar los archivos del servidor)
#
<IfModule mod_alias.c>
  Redirect 301 /1dnest/product.html  https://1d-solutions.com/es/product.html
  Redirect 301 /1dnest/indice.html   https://1d-solutions.com/es/index.html
  Redirect 301 /1dnest/1DNest.html   https://1d-solutions.com/es/index.html

  # library.html era la DLL CutLib y contac.html el contacto: cada una a su
  # página equivalente, no a la portada.
  Redirect 301 /1dnest/library.html  https://1d-solutions.com/es/product.html
  Redirect 301 /1dnest/contac.html   https://1d-solutions.com/es/contact.html

  # Resto de /1dnest/ a la home española. Al final, porque gana la primera regla
  # que encaja y las de arriba deben poder ganar.
  #

  # lo que sobra de la ruta, así que /1dnest/otra.html acabaría en
  # ".../es/index.htmlotra.html".

</IfModule>

# ▼▼▼ INICIO AÑADIDO 2026-08-15 — 4. Archivos que no deben servirse nunca ▼▼▼
#
# Red de seguridad para el día en que una subida a mano cuele una carpeta de
# más. Ninguno de estos archivos existe hoy en public_html.
#
# El patrón "^\." hace falta: este hosting bloquea .env y .git por nombre, no
# por patrón, y un .DS_Store filtraría el listado del directorio.
#
# No alcanza a .html, .css, .js, .svg, .webp ni a los .exe/.zip de /_downloads/.
#
# vault: postura-de-seguridad-del-sitio · publicacion-del-sitio-1d
#
<FilesMatch "(^\.|composer\.(json|lock)|package(-lock)?\.json|gulpfile\.js|netlify\.toml|\.(sh|bak|sql|log|md)$)">
  # "Require" es sintaxis de Apache 2.4: en 2.2 provoca un 500 en TODO el sitio,
  # no sólo en los archivos afectados. De ahí el guardia IfModule.
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order allow,deny
    Deny from all
  </IfModule>
</FilesMatch>

# Listado de directorios desactivado explícitamente, para no depender de la
# configuración global de GoDaddy.
<IfModule mod_autoindex.c>
  Options -Indexes
</IfModule>
#
# ▲▲▲ FIN AÑADIDO — 4. Archivos que no deben servirse nunca ▲▲▲


# ▼▼▼ INICIO AÑADIDO 2026-08-26 — 5. SEO: www → sin-www, y restos de /pt/ y /esp/ ▼▼▼
#
# vault: plan-seo-organico · sitio-zombi-1dnest
#
# a) El dominio canónico es sin www. La regla de HTTPS de arriba conserva el host
#    (%{HTTP_HOST}), así que no quita el www: sin esta regla, www sirve el sitio
#    completo con 200 y Google ve un duplicado. El certificado cubre www.
#    Una petición http://www hace dos saltos (→ https://www → apex).
#
RewriteCond %{HTTP_HOST} ^www\.1d-solutions\.com$ [NC]
RewriteRule ^(.*)$ https://1d-solutions.com/$1 [L,R=301]

# b) Restos portugueses del sitio anterior, hoy en 404. 301 al equivalente vivo
#    para conservar el enlazado externo. El bloque 3 sólo cubre /1dnest/.
#
<IfModule mod_alias.c>
  Redirect 301 /pt/indice.html   https://1d-solutions.com/pt/index.html
  Redirect 301 /pt/library.html  https://1d-solutions.com/pt/index.html
</IfModule>

# c) /esp/ era la versión española anterior a /es/. Devuelve 404 desde julio de
#    2025 y sigue recibiendo clics de búsqueda y de pago.
#
<IfModule mod_alias.c>
  Redirect 301 /esp/index.html    https://1d-solutions.com/es/index.html
  Redirect 301 /esp/product.html  https://1d-solutions.com/es/product.html
  Redirect 301 /esp/contact.html  https://1d-solutions.com/es/contact.html
  RedirectMatch 301 ^/esp/?$      https://1d-solutions.com/es/index.html
</IfModule>
#
# ▲▲▲ FIN AÑADIDO — 5. SEO ▲▲▲


# ▼▼▼ INICIO AÑADIDO 2026-08-27 — 6. Rutas antiguas con tráfico medido ▼▼▼
#
# Rutas del sitio anterior que devuelven 404 y que GA4 muestra recibiendo visitas
# —por Bing, enlace directo y clics de pago, además de Google—, y que llevaban
# enlaces a ShareIt, la pasarela que ya no cobra. El destino de cada una se
# eligió por su idioma y su tema, no al bulto.
#
# vault: auditoria-urls-legacy-y-pasarela (cifras y método) · t0020 (subir este
# archivo y verificarlo) · t0013 (borrar los archivos) · t0014 (el PAD)
#
<IfModule mod_alias.c>
  # La DLL CutLib, en la raíz y en sus copias antiguas
  Redirect 301 /library.html          https://1d-solutions.com/product.html
  Redirect 301 /z-OLD/library.html    https://1d-solutions.com/product.html
  Redirect 301 /z-OLD/product.html    https://1d-solutions.com/product.html
  Redirect 301 /z-OLD/contac.html     https://1d-solutions.com/es/contact.html
  Redirect 301 /z-OLD/index.html      https://1d-solutions.com/index.html

  # Restos en la raíz
  Redirect 301 /contac.html           https://1d-solutions.com/contact.html
  Redirect 301 /indice.html           https://1d-solutions.com/es/index.html

  # NO AÑADIR una regla para /index.htm. Se probó el 2026-08-27 y rompe la
  # portada: al pedir «/», mod_dir busca el índice del directorio probando
  # index.htm antes que index.html, y esa subpetición interna activa la regla,
  # así que «/» acaba devolviendo 301 en vez de servir la página. Sólo afecta a
  # la raíz —«/es/» genera /es/index.htm, que una regla anclada en ^/index no
  # captura—, lo que hace el fallo fácil de pasar por alto.
  # /index.htm no tenía ningún tráfico medido, así que no se pierde nada.
  #
  # El ancla «^...$» sigue siendo obligatoria aquí: "Redirect 301 /product.htm"
  # casaría por prefijo con /product.html y la mandaría a /product.htmll.
  RedirectMatch 301 ^/product\.htm$   https://1d-solutions.com/product.html

  # Borradores en inglés
  Redirect 301 /index_EX.html         https://1d-solutions.com/index.html
  Redirect 301 /en/index1.html        https://1d-solutions.com/index.html
  Redirect 301 /en/product1.html      https://1d-solutions.com/product.html

  # Página de venta con enlaces a ShareIt que quedó dentro del canal de
  # descargas. La 301 la neutraliza aunque el archivo siga en el servidor:
  # mod_alias resuelve antes de mapear al sistema de archivos.
  Redirect 301 /_downloads/entp/product.html  https://1d-solutions.com/product.html

  # Restos de /pt/ que no cubre el bloque 5b. El .htm va anclado por lo mismo
  # que arriba: si no, casaría con /pt/indice.html.
  RedirectMatch 301 ^/pt/indice\.htm$ https://1d-solutions.com/pt/index.html
  Redirect 301 /pt/contac.html        https://1d-solutions.com/pt/contact.html
  Redirect 301 /pt/page1.html         https://1d-solutions.com/pt/index.html

  # Carpetas muertas completas. Al final, porque gana la primera regla que
  # encaja. RedirectMatch por el mismo motivo que en el bloque 3.
  RedirectMatch 301 ^/z-OLD/    https://1d-solutions.com/index.html
  RedirectMatch 301 ^/nuevo/    https://1d-solutions.com/es/index.html
  RedirectMatch 301 ^/new/      https://1d-solutions.com/index.html
  RedirectMatch 301 ^/prueba/   https://1d-solutions.com/index.html
  RedirectMatch 301 ^/J1DN/     https://1d-solutions.com/index.html
</IfModule>
#
# ▲▲▲ FIN AÑADIDO — 6. Rutas antiguas con tráfico medido ▲▲▲
