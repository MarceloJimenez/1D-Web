# ATENCIÓN — ESTO NO SUSTITUYE AL .htaccess QUE YA HAY
#
# El servidor ya tiene un .htaccess con el forzado de HTTPS (se comprobó el
# 2026-08-15: la redirección de http a https conserva la ruta y los parámetros,
# que es la firma de mod_rewrite, no de una redirección del hosting). Si se
# reemplaza el archivo, el sitio deja de forzar HTTPS.
#
# Este bloque se PEGA AL FINAL del .htaccess existente, sin borrar nada.
#
# Todas las directivas van dentro de <IfModule>. Si un módulo no está cargado,
# Apache se salta el bloque en lugar de devolver un error. No hay ninguna
# directiva suelta que pueda tumbar el sitio por sí misma.
#
# Instalación y marcha atrás: ver README.md en esta misma carpeta.

# -----------------------------------------------------------------------------
# 1. Cabeceras de seguridad
# -----------------------------------------------------------------------------
#
# Hasta 2026-08-15 la respuesta del sitio no traía ni una. El netlify.toml de la
# raíz del repositorio declara algunas, pero el sitio no está en Netlify y nadie
# las aplicaba — es justamente la prueba de que ese archivo no se usa.
#
# Ninguna de estas cambia el contenido ni las rutas: sólo añaden cabeceras a la
# respuesta. Es la parte de menos riesgo de todo el archivo.
#
<IfModule mod_headers.c>

  # Obliga a HTTPS durante un año. El sitio ya redirige a HTTPS y el certificado
  # de GoDaddy se renueva solo. Deliberadamente SIN includeSubDomains y SIN
  # preload:
  #   - includeSubDomains alcanzaría a cualquier subdominio futuro, incluido uno
  #     de pruebas sin certificado, y lo dejaría inaccesible.
  #   - preload es prácticamente irreversible (hay que pedir la baja a Google y
  #     esperar meses).
  # Es la única directiva con efecto persistente en el navegador del visitante:
  # quien la reciba usará HTTPS durante un año aunque mañana se quite de aquí.
  Header always set Strict-Transport-Security "max-age=31536000"

  # Impide que el navegador adivine el tipo de un archivo ignorando el
  # Content-Type. Sin esto, un .txt subido por error puede acabar ejecutándose
  # como HTML.
  Header always set X-Content-Type-Options "nosniff"

  # Nadie puede meter el sitio en un iframe salvo el propio sitio. Evita
  # clickjacking. Ningún tercero embebe 1d-solutions.com: el checkout de PayPro
  # es un enlace que abre su dominio, no un iframe.
  Header always set X-Frame-Options "SAMEORIGIN"

  # Al salir hacia otro dominio se envía sólo el origen, no la URL completa.
  # Dentro del sitio se sigue enviando entera, que es lo que Analytics necesita.
  Header always set Referrer-Policy "strict-origin-when-cross-origin"

  # El sitio no usa cámara, micrófono, geolocalización ni pagos del navegador.
  # Declararlo vacío impide que se los pida cualquier script inyectado — o
  # cualquier etiqueta que alguien añada a GTM sin pensarlo.
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=(), usb=()"

  # Oculta la versión de PHP si el hosting la anuncia.
  Header always unset X-Powered-By

</IfModule>


# -----------------------------------------------------------------------------
# 2. Content-Security-Policy — EN MODO INFORME, a propósito
# -----------------------------------------------------------------------------
#
# Report-Only NO bloquea nada: sólo escribe las violaciones en la consola del
# navegador. Está así por dos razones concretas:
#
#   a) Todo el HTML lleva scripts en línea (el snippet de GTM, loadNonCritical,
#      addToCartSingle, el document.write del año). Una CSP sin 'unsafe-inline'
#      los mataría todos, y con 'unsafe-inline' la protección contra XSS es
#      floja de todas formas. Lo que aporta de verdad es la lista de dominios.
#
#   b) GTM inyecta etiquetas en tiempo de ejecución. Añadir una etiqueta nueva
#      en GTM-TGXKJ8P puede introducir un dominio que no esté en esta lista, y
#      en modo bloqueo eso rompe la medición sin que nadie lo note hasta pasadas
#      semanas — que es exactamente el modo de fallo que este proyecto persigue.
#
# Va en UNA SOLA LÍNEA aunque sea larguísima, a propósito. Apache admite partir
# líneas con "\" al final, pero si al pegar en el editor de cPanel queda un
# espacio detrás de la barra, Apache falla y el sitio devuelve 500. Una línea
# larga no tiene ese problema. NO la partas al copiarla.
#
# CÓMO PASARLA A BLOQUEO, cuando toque: navegar el sitio entero (3 páginas × 9
# idiomas, más el formulario de contacto y un clic de compra) con la consola
# abierta, apuntar cada violación, añadir los dominios que falten, y sólo
# entonces quitar el sufijo "-Report-Only" del nombre de la cabecera.
#
<IfModule mod_headers.c>
  Header always set Content-Security-Policy-Report-Only "default-src 'self'; script-src 'self' 'unsafe-inline' https://www.googletagmanager.com https://www.google-analytics.com https://googleads.g.doubleclick.net https://www.googleadservices.com https://www.google.com https://web3forms.com https://*.hcaptcha.com https://hcaptcha.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com https://*.hcaptcha.com; font-src 'self' https://fonts.gstatic.com data:; img-src 'self' data: https://www.googletagmanager.com https://www.google-analytics.com https://www.google.com https://googleads.g.doubleclick.net https://td.doubleclick.net https://pagead2.googlesyndication.com; connect-src 'self' https://www.google-analytics.com https://region1.google-analytics.com https://analytics.google.com https://www.googletagmanager.com https://api.web3forms.com https://pagead2.googlesyndication.com https://*.hcaptcha.com; frame-src https://www.googletagmanager.com https://td.doubleclick.net https://*.hcaptcha.com https://hcaptcha.com; form-action 'self' https://api.web3forms.com https://store.payproglobal.com; base-uri 'self'; object-src 'none'; frame-ancestors 'self'"
</IfModule>


# -----------------------------------------------------------------------------
# 3. Las páginas de /1dnest/
# -----------------------------------------------------------------------------
#
# Cuatro páginas de 2021-2024 que seguían respondiendo 200, con analítica vieja
# (UA-209685-2, una propiedad Universal Analytics que ya ni recoge datos) y
# enlaces a ShareIt, la pasarela anterior a PayPro. Además cargaban
# https://cdn.ywxi.net/js/1.js — el sello de TrustedSite, 42 KB de JavaScript de
# terceros con ejecución completa, en páginas que no mantiene nadie.
#
# 301 en vez de borrarlas: conserva el poco enlazado externo que tengan y evita
# 404 en Search Console. Van al español porque el contenido estaba en español.
#
# Esto SÍ cambia el comportamiento de URLs. Son las cuatro de /1dnest/ y nada
# más; ninguna página viva del sitio pasa por aquí.
#
<IfModule mod_alias.c>
  Redirect 301 /1dnest/product.html  https://1d-solutions.com/es/product.html
  Redirect 301 /1dnest/indice.html   https://1d-solutions.com/es/index.html
  Redirect 301 /1dnest/library.html  https://1d-solutions.com/es/index.html
  Redirect 301 /1dnest/1DNest.html   https://1d-solutions.com/es/index.html

  # Cualquier otra cosa que quede colgando bajo /1dnest/ va a la home en español.
  # Va al final porque gana la primera regla que encaja: las cuatro de arriba
  # tienen que poder ganar.
  #
  # RedirectMatch y no Redirect, y no es un detalle de estilo: "Redirect" casa
  # por prefijo y AÑADE al destino lo que sobra de la ruta, así que
  # /1dnest/otra.html habría acabado en ".../es/index.htmlotra.html". Se vio
  # probándolo contra un Apache local el 2026-08-15. RedirectMatch sin grupos de
  # captura usa el destino tal cual.
  RedirectMatch 301 ^/1dnest/ https://1d-solutions.com/es/index.html
</IfModule>


# -----------------------------------------------------------------------------
# 4. Denegar acceso a archivos que no deberían servirse nunca
# -----------------------------------------------------------------------------
#
# Hoy ninguno de estos existe en public_html — se comprobó el 2026-08-15 y todos
# daban 404. Esto es una red de seguridad para el día en que alguien suba una
# carpeta de más: el despliegue es arrastrar archivos a mano, y ya pasó una vez
# que el sitio entero acabara duplicado en public_html/theme/.
#
# El patrón no toca .html, .css, .js, .svg, .webp ni nada que el sitio sirva.
# Sí bloquea .zip: si alguna vez se quiere ofrecer un zip para descarga, hay que
# quitarlo de aquí. Los .exe de /_downloads/ no están en la lista y siguen
# funcionando.
#
<FilesMatch "(^\.|composer\.(json|lock)|package(-lock)?\.json|gulpfile\.js|netlify\.toml|\.(env|sh|bak|sql|log|zip|md)$)">
  # El guardia IfModule no es decorativo: "Require" es sintaxis de Apache 2.4 y
  # en 2.2 provoca un 500 en TODO el sitio, no sólo en los archivos afectados.
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order allow,deny
    Deny from all
  </IfModule>
</FilesMatch>

# El listado de directorios ya venía desactivado en este hosting; se deja
# explícito para que no dependa de la configuración global de GoDaddy.
<IfModule mod_autoindex.c>
  Options -Indexes
</IfModule>
